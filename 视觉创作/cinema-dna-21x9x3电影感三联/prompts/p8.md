## 完整 PIL 拼接代码

```python
from PIL import Image, ImageDraw, ImageFont

# 加载 3 张 triptych shots
img1 = Image.open("shot1.png").convert("RGB")  # 建立世界
img2 = Image.open("shot2.png").convert("RGB")  # 建立关系
img3 = Image.open("shot3.png").convert("RGB")  # 留下余韵

# 标准化到 21:9 比例
target_w, target_h = 1024, 432  # 21:9 = 2.37

def resize_21_9(img):
    w, h = img.size
    target_ratio = target_w / target_h
    if w / h > target_ratio:
        new_h = int(w / target_ratio)
        top = (new_h - h) // 2
        img = img.crop((0, top, w, top + h))
    else:
        new_w = int(h * target_ratio)
        left = (new_w - w) // 2
        img = img.crop((left, 0, left + w, h))
    return img.resize((target_w, target_h), Image.LANCZOS)

img1, img2, img3 = map(resize_21_9, [img1, img2, img3])

# 1.25:0.9:0.85 比例（强调首镜头）
ratio1, ratio2, ratio3 = 1.25, 0.9, 0.85
total = ratio1 + ratio2 + ratio3
w1 = int(target_w * ratio1 / total)
w2 = int(target_w * ratio2 / total)
w3 = int(target_w * ratio3 / total)

img1 = img1.resize((w1, target_h), Image.LANCZOS)
img2 = img2.resize((w2, target_h), Image.LANCZOS)
img3 = img3.resize((w3, target_h), Image.LANCZOS)

# 拼接 + 8px 黑色间隔
gap = 8
total_w = w1 + w2 + w3 + gap * 2
canvas = Image.new("RGB", (total_w, target_h), (0, 0, 0))
x = 0
canvas.paste(img1, (x, 0))
x += w1 + gap
canvas.paste(img2, (x, 0))
x += w2 + gap
canvas.paste(img3, (x, 0))

# 可选：加文字标注
draw = ImageDraw.Draw(canvas)
font = ImageFont.truetype("/System/Library/Fonts/Helvetica.ttc", 16)
draw.text((10, 10), "cinema-dna · 1.25:0.9:0.85", fill=(255, 200, 100), font=font)

# 保存
canvas.save("triptych-final.jpg", "JPEG", quality=85, optimize=True)
print(f"Done: {canvas.size}")
```

## 工具对比

| 维度 | 普通 AI 配图 | cinema-dna | scroll-world | dashiai-ppt |
|---|---|---|---|---|
| 输出 | 1 张竖图 | 3 张 21:9 横向叙事 | 视频连续飞行 | 22 幕讲解 |
| 核心 | "电影感滤镜" | 镜头成立原因 | seam frame-identical | 横屏适配 |
| 工具 | matrix × 1 | matrix × 3 + PIL | matrix × N + ffmpeg + scrub | matrix × 1 + PIL |
| 耗时 | 30s | 5-10 min | 40-60 min | 30-60 min |
| 适用 | 随手配图 | 高级感主题 | 品牌沉浸 landing | 演讲/汇报 |
| 不适用 | 高品质场景 | 纯科普/数据 | 静态图 | 视频/动画 |
